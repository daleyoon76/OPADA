```python
# 저장 경로 예: /tmp/r1_exit13.py · 실행: python3 /tmp/r1_exit13.py tests/answer_set/13.json
import json,re,sys
d=json.load(open(sys.argv[1] if len(sys.argv)>1 else 'tests/answer_set/13.json'))
n=lambda s: re.sub(r'\s+','',s)
heads=('8.낙찰자','2)','3)','4)','5)','※각종','6)','7)','8)','9)')
eight=[i for i in d['items'] if i['evidence']['source']=='13_1.txt' and n(i['evidence']['quote']).startswith(heads) and '14일' not in n(i['evidence']['quote']) and '사용개시일' not in n(i['evidence']['quote'])]
cheong=[i for i in d['items'] if '청렴' in i['name']]
biz=[i for i in d['items'] if '사업자등록증' in i['name']]
bad=[i['name'] for i in eight+cheong if i['stage']!='계약시']
print(f"8항 대상 {len(eight)}건(하한 11) · stage≠계약시 {len(bad)} · 청렴 {len(cheong)} · 사업자등록증 {len(biz)}")
ok = len(eight)>=11 and not bad and len(cheong)==1 and len(biz)==1
print('PASS' if ok else 'FAIL'); sys.exit(0 if ok else 1)
```
